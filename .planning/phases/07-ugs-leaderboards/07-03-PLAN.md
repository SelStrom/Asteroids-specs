Добавить метод SubmitScoreCoroutine() как private IEnumerator:
```csharp
private IEnumerator SubmitScoreCoroutine(string playerName, int score)
{
    if (_ugsService == null)
    {
        _gameOverScreen.NotifySubmitError("UGS сервис недоступен.");
        yield break;
    }

    // D-02: сохранить имя в PlayerPrefs
    if (!string.IsNullOrEmpty(playerName))
    {
        PlayerPrefs.SetString("PlayerName", playerName);
    }

    var task = _ugsService.SubmitScoreAsync(playerName, score);
    while (!task.IsCompleted)
    {
        yield return null;
    }

    if (task.IsFaulted)
    {
        // LEAD-05: показать ошибку
        _gameOverScreen.NotifySubmitError("Ошибка отправки. Проверьте соединение.");
        Debug.LogWarning("[Application] Ошибка submit score: " + task.Exception?.Message);
    }
    else
    {
        // D-03: успех — отключить кнопку, перейти к LeaderboardScreen
        _gameOverScreen.NotifySubmitSuccess();
        // Автоматически открыть Leaderboard после успешного submit
        yield return new WaitForSeconds(0.5f); // краткая пауза для визуальной обратной связи
        OnLeaderboardOpen();
    }
}
```

```csharp
private IEnumerator InitUgsCoroutine()
{
    if (_ugsService == null) { yield break; }
    var task = _ugsService.InitializeAsync();
    yield return RunAsync(task);
    if (task.IsFaulted)
    {
        // LEAD-05: UGS недоступен — кнопки лидерборда остаются выключены
        Debug.LogWarning("[UGS] Инициализация не удалась: " + task.Exception?.Message);
    }
    else
    {
        Debug.Log("[UGS] Инициализация успешна. Игрок залогинен анонимно.");
    }
}

// D-09: Coroutine-обёртка для Task (паттерн из RESEARCH.md)
public static IEnumerator RunAsync(System.Threading.Tasks.Task task)
{
    while (!task.IsCompleted)
    {
        yield return null;
    }
    if (task.IsFaulted && task.Exception != null)
    {
        throw task.Exception.InnerException ?? task.Exception;
    }
}
```
