```csharp
#if UNITY_EDITOR
using SelStrom.Asteroids;
using SelStrom.Asteroids.Configs;
using TMPro;
using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine;
using UnityEngine.UI;

namespace SelStrom.Asteroids.Editor
{
    public static class Phase7Setup
    {
        private const string ScenePath = "Assets/Scenes/Main.unity";
        private const string GameDataPath = "Assets/Media/configs/GameData.asset";

        [MenuItem("Asteroids/Setup Phase 7 Assets")]
        public static void SetupAll()
        {
            UpdateGameOverScreen();
            UpdateLeaderboardScreen();
            UpdateGameDataLeaderboardId();
            AssetDatabase.SaveAssets();
            AssetDatabase.Refresh();
            Debug.Log("[Phase7Setup] Готово. Все Phase 7 assets настроены.");
        }
```

**Метод UpdateGameOverScreen():**
- Открыть сцену Main.unity
- Найти GameOverView через `Resources.FindObjectsOfTypeAll<GameOverView>()`
- Найти GameObject "GameOverScreen" (родитель GameOverView)
- Добавить TMP_InputField "player_name_input":
  - Создать дочерний GameObject с RectTransform (width=240, height=40)
  - Добавить Image компонент (цвет #1A1A1A)
  - Добавить TMP_InputField компонент
  - Установить characterLimit=16, placeholder text="Введите имя"
  - Разместить над кнопкой Submit Score (anchoredPosition примерно (0, -Y) в зависимости от текущего layout)
  - Использовать якоря (0.5f, 0.5f) center
- Добавить TextMeshProUGUI "error_text":
  - Создать дочерний GameObject
  - TMP fontSize=14, color=new Color(1f, 0.267f, 0.267f, 1f) (#FF4444)
  - alignment=Center
  - text="" по умолчанию
  - SetActive(false)
- Назначить через SerializedObject:
  - `soView.FindProperty("_playerNameInput").objectReferenceValue = inputField`
  - `soView.FindProperty("_errorText").objectReferenceValue = errorText`
- MarkSceneDirty + SaveScene

**Метод UpdateLeaderboardScreen():**
- Открыть сцену Main.unity
- Найти LeaderboardView через `Resources.FindObjectsOfTypeAll<LeaderboardView>()`
- Найти существующий "placeholder_text" — скрыть его (`SetActive(false)`) но не удалять (используется как "Загрузка...")
- Создать 10 TextMeshProUGUI строк "_entry_0" .. "_entry_9":
  - Каждая: fontSize=18, alignment=Left, color=white
  - Расположение: вертикальный стек начиная с Y=-160 (под заголовком), шаг -30px (sm=8px + высота строки 22px)
  - SetActive(false) по умолчанию (Bind() их активирует)
- Создать Image "_separator":
  - width = 500, height = 1px
  - color = white
  - Расположение: Y = после 10 строки + xl=32px = примерно Y=-490
- Создать TextMeshProUGUI "_player_entry":
  - fontSize=18, alignment=Left, color=white
  - Расположение: Y после separator + lg=24px
  - SetActive(false) по умолчанию
- Создать TextMeshProUGUI "_error_text":
  - fontSize=14, color=#FF4444 (1f, 0.267f, 0.267f, 1f)
  - alignment=Center
  - SetActive(false) по умолчанию
- Назначить через SerializedObject:
  - `soView.FindProperty("_entryTexts").arraySize = 10`
  - Для каждого i: `soView.FindProperty("_entryTexts").GetArrayElementAtIndex(i).objectReferenceValue = entryTexts[i]`
  - `soView.FindProperty("_playerEntryText").objectReferenceValue = playerEntry`
  - `soView.FindProperty("_errorText").objectReferenceValue = errorText`
- MarkSceneDirty + SaveScene

**Метод UpdateGameDataLeaderboardId():**
- Загрузить GameData.asset: `AssetDatabase.LoadAssetAtPath<GameData>(GameDataPath)`
- Через SerializedObject: проверить что LeaderboardId = "asteroids_highscores"
- Если пустое — установить значение по умолчанию
- EditorUtility.SetDirty(gameData)

Для TitleScreenView._leaderboardButton — активация происходит автоматически через Application.Connect() в Runtime (из Plan 03 Task 1: interactable = OnLeaderboardClicked != null). В Editor-режиме кнопка остаётся в том состоянии в котором была сохранена — Phase7Setup может принудительно установить interactable=true:
- Найти TitleScreenView через `Resources.FindObjectsOfTypeAll<TitleScreenView>()`
- Через SerializedObject найти "_leaderboardButton" и установить interactable=true на Button компоненте напрямую: `button.interactable = true; EditorUtility.SetDirty(button);`
